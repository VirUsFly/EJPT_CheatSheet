```
#!/bin/bash

TARGET="192.150.72.3"
WORDLIST="/root/Desktop/wordlists/shares.txt"

if [ ! -f "$WORDLIST" ]; then
    echo "[-] Wordlist not found: $WORDLIST"
    exit 1
fi

echo "[*] Starting SMB share enumeration on $TARGET"
echo

while read -r SHARE; do
    if [ -n "$SHARE" ]; then
        echo "[*] Testing share: $SHARE"
        smbclient -N "//$TARGET/$SHARE" -c "ls" >/dev/null 2>&1
        if [ $? -eq 0 ]; then
            echo "[+] Accessible share found: $SHARE"
        else
            echo "[-] Not accessible: $SHARE"
        fi
    fi
done < "$WORDLIST"

echo
echo "[*] Scan completed."
```
